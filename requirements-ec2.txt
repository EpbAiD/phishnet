# Minimal deps for the EC2 daemon + extraction script.
#
# Do NOT put the full ML stack (catboost/xgboost/lightgbm/scikit-learn/shap)
# here. Those are for training, which runs on GitHub Actions runners with
# their own newer Python. The EC2 daemon only does DNS/WHOIS extraction and
# S3 accumulation — it doesn't need any ML libs.
#
# Chosen against Amazon Linux 2023 (ships Python 3.9 by default). Every
# version here was cross-checked on pypi to be the newest release whose
# `requires_python` accepts 3.9. If we ever bump the EC2 AMI to something
# with Python 3.10+, we can bump all of these to the latest.
#
# ipwhois 1.3.0 (Aug 2024) dropped the old `dnspython<=2.0.0` cap, so we
# can finally use a modern dnspython here.

boto3==1.42.97
pandas==2.3.3
python-whois==0.9.6
dnspython==2.7.0
ipwhois==1.3.0
tldextract==5.3.0
requests==2.32.5
# Postgres client for upserting batches into RDS. Replaces the CSV-in-S3
